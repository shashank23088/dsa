vector<vector<int>> prePostInorder(Node* root) {
    vector<vector<int>> answer;
    if(!root) {return answer;}

    vector<int> preorder, postorder, inorder;
    queue<pair<Node*, int>> q;
    q.push({root, 1});

    while(!q.empty()) {
        Node* curr_node = q.front().first;
        Node* curr_freq = q.front().second;
        q.pop();

        if(curr_freq == 1) {
            preorder.emplace_back(curr_node -> val);
            q.push({curr_node, curr_freq + 1});
            if(curr_node -> left) {
                q.push({curr_node -> left, 1});
            }
        }

        else if(curr_freq == 2) {
            inorder.emplace_back(curr_node -> val);
            q.push({curr_node, curr_freq + 1});
            if(curr_node -> right) {
                q.push({curr_node -> right, 1});
            }
        }

        else {
            postorder.emplace_back(curr_node -> val);
        }
    }

    answer.emplace_back(preorder);
    answer.emplace_back(inorder);
    answer.emplace_back(postorder);

    return answer;
}
